#!/sbin/sh
#********************************************************************
#*
#*   シェル名      : 変換後新製品ファミリー情報ロード(作表区分付加)
#*
#*   シェルＩＤ    : M15SHINFMLD
#*
#*   機能          : 変換後新製品ファミリー情報のロード(作表区分付加)
#*
#*   作成日        : 2002/02/19               Created  by  D.S-M.T
#*
#********************************************************************
SHELLID=M15SHINFM;export SHELLID
HOME=/home/hanbai
#LOG=$HOME/LOGBOX;export LOG
LOG=/home/hanbai/log/M15SINSEI.lst;export LOG
PATH=/oracle/app/oracle/product/8.1.7/bin:/usr/bin:/usr/sbin:/etc:/home/hanbai:/home/hanbai/infile/:/opt/hulft/bin:.:/usr/local/bin
HZ=100
TZ=:Japan
TERM=vt100
SHELL=/bin/sh
TIMEOUT=0
PWD=/home/hanbai
ORACLE_HOME=/oracle/app/oracle/product/8.1.7
ORACLE_SID=hanbai
ORACLE_TERM=sun5
LANG=japan
LANGUAGE=Japanese_Japan.JA16EUC
NLS_LANG=Japanese_Japan.JA16EUC
NLS_LANGUAGE=Japanese
NLS_TERRITORY=Japan
HULPATH=/opt/hulft/etc
INDATA=/home/hanbai/infile/A0003103.dat.csv
INDEX1=/home/hanbai/sql/C_PK_00_15SHINFM.sql
MSG=/home/hanbai/msg/SINSEI_ERR.log;export MSG

export  HOME PATH HZ TZ TERM SHELL TIMEOUT PWD ORACLE_HOME ORACLE_SID ORACLE_TERM LANG LANGUAGE NLS_LANG NLS_LANGUAGE NLS_TERRITORY HULPATH INDATA

#STARTメッセージ

##rm $HOME/log/$SHELLID.bad
##rm $LOG

echo "****** "$SHELLID":START ******" >> $LOG

date >> $LOG
echo "****** 新製品ファミリーテーブル制約禁止  START!!! *****" >> $LOG
PROGID=drop_constraint;export PROGID
sqlplus ouriage/ouriage <<!
    alter table TB_00_15SHINFM disable constraint PK_00_15SHINFM;
    exit SQL.SQLCODE
!
STATUS=$?;export STATUS
case $STATUS in

0)
      echo $PROGID":正常終了" >> $LOG
      ;;
*)
      echo $PROGID":ＳＱＬ検出エラー エラーコード("$STATUS")" >> $LOG
      date >> $LOG

      echo "総合ＤＢ 販売系（新製品）：異常終了" > $MSG
      date >> $MSG
      echo "新製品ファミリーテーブル制約削除：異常終了" >> $MSG
      echo $SHELLID"."$PROGID":ＳＱＬ検出エラー エラーコード("$STATUS")" >> $MSG

      exit $STATUS
      ;;
esac

date >> $LOG
echo "****** 新製品ファミリー情報ロード     START!!! *****" >> $LOG
PROGID=SQLload;export PROGID
sqlldr ouriage/ouriage, $HOME/loadctl/$SHELLID.ctl,$HOME/log/$SHELLID.log,$HOME/log/$SHELLID.bad,DATA=$INDATA,DISCARD=$HOME/log/$SHELLID.dis, errors=99999, direct=true
STATUS=$?;export STATUS
case $STATUS in
0)
      echo $PROGID":正常終了" >> $LOG
      ;;
*)
      echo $PROGID":ＳＱＬ検出エラー エラーコード("$STATUS")" >> $LOG
      date >> $LOG

      echo "総合ＤＢ 販売系（新製品）：異常終了" > $MSG
      date >> $MSG
      echo "新製品ファミリーテーブルロード：異常終了" >> $MSG
      echo $SHELLID"."$PROGID":ＳＱＬ検出エラー エラーコード("$STATUS")" >> $MSG

      exit $STATUS
      ;;
esac

date >> $LOG
echo "****** 新製品ファミリーテーブル制約開始  START!!! *****" >> $LOG
PROGID=create_constraint;export PROGID
sqlplus ouriage/ouriage <<!
    alter table TB_00_15SHINFM enable constraint PK_00_15SHINFM;   
    exit SQL.SQLCODE
!
STATUS=$?;export STATUS
case $STATUS in

0)
      echo $PROGID":正常終了" >> $LOG
      ;;
*)
      echo $PROGID":ＳＱＬ検出エラー エラーコード("$STATUS")" >> $LOG
      date >> $LOG

      echo "総合ＤＢ 販売系（新製品）：異常終了" > $MSG
      date >> $MSG
      echo "新製品ファミリーテーブル制約付加：異常終了" >> $MSG
      echo $SHELLID"."$PROGID":ＳＱＬ検出エラー エラーコード("$STATUS")" >> $MSG

      exit $STATUS
      ;;
esac

echo "新製品ファミリーテーブル作成：正常終了" > $MSG
date >> $MSG

date >> $LOG

##rm $HOME/infile/sinseihin.dat

echo "****** "$SHELLID":END ********" >> $LOG
